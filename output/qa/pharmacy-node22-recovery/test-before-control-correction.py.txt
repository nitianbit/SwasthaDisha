SOURCE = Path(__file__).with_name('recover-node22.py').read_text()

def load(source=SOURCE):
    module = types.ModuleType('offline_node22_operator')
    module.__file__ = str(Path(__file__).with_name('recover-node22.py'))
    exec(compile(source, module.__file__, 'exec'), module.__dict__)
    return module

op = load()
real_network_attempts = []
def deny_network(*args, **kwargs):
    real_network_attempts.append(True)
    raise AssertionError('REAL_NETWORK_FORBIDDEN')

class Response:
    def __init__(self, data, headers=None, status=200):
        self.status = status
        self.data = io.BytesIO(data)
        self.headers = [('Content-Length', str(len(data)))] if headers is None else headers
    def getheaders(self):
        return self.headers
    def read(self, amount):
        return self.data.read(amount)

class Connection:
    def __init__(self, response):
        self.response, self.closed, self.requests, self.sock = response, False, [], self
    def connect(self):
        pass
    def settimeout(self, value):
        assert value == 5
    def request(self, method, path, headers):
        self.requests.append((method, path, headers))
    def getresponse(self):
        return self.response
    def close(self):
        self.closed = True

def archive_bytes(kind='regular', duplicate=False, extra_link=False):
    output = io.BytesIO()
    with tarfile.open(fileobj=output, mode='w') as archive:
        if extra_link:
            link = tarfile.TarInfo('node-v22.23.4-darwin-arm64/bin/npm')
            link.type, link.linkname = tarfile.SYMTYPE, '../../outside'
            archive.addfile(link)
        item = tarfile.TarInfo(op.MEMBER if kind != 'missing' else 'other/bin/node')
        data = b'fixed fake binary; never executed'
        if kind == 'symlink':
            item.type, item.linkname = tarfile.SYMTYPE, '/outside'
            archive.addfile(item)
        else:
            item.size = len(data)
            archive.addfile(item, io.BytesIO(data))
            if duplicate:
                archive.addfile(item, io.BytesIO(data))
    return gzip.compress(output.getvalue()), data

class OfflineChecks(unittest.TestCase):
    def setUp(self):
        self.workspace = tempfile.TemporaryDirectory(prefix='sd-node22-offline-', dir='/private/tmp')
        self.root = Path(self.workspace.name)
        self.stack = contextlib.ExitStack()
    def tearDown(self):
        self.stack.close()
        self.workspace.cleanup()
    def fd(self, name):
        fd = os.open(self.root/name, os.O_RDWR | os.O_CREAT | os.O_EXCL, 0o600)
        self.stack.callback(os.close, fd)
        return fd
    def invoke_download(self, response, cap=1024):
        connection = Connection(response)
        result = op.download(op.RELEASE+op.ARCHIVE, cap, self.fd('download'), lambda:connection)
        self.assertTrue(connection.closed)
        self.assertEqual(connection.requests[0][0:2], ('GET', op.RELEASE+op.ARCHIVE))
        return result
    def invoke_unpack(self, data, binary, **options):
        source, target = self.fd('archive'), self.fd('node')
        os.write(source, data)
        return op.unpack(source, target, expected_archive=hashlib.sha256(data).hexdigest(), expected_binary=hashlib.sha256(binary).hexdigest(), **options)
    def core(self, fail=False):
        archive, binary = archive_bytes(extra_link=True)
        digest = hashlib.sha256(archive).hexdigest()
        responses = [Response((digest+'  '+op.ARCHIVE+'\n').encode()), Response(archive)]
        calls = []
        def factory():
            calls.append(True)
            if fail:
                raise op.Refusal('first download failure')
            return Connection(responses.pop(0))
        with patch.object(op, 'ARCHIVE_SHA', digest), patch.object(op, 'BINARY_SHA', hashlib.sha256(binary).hexdigest()):
            result = op.recover(self.root, factory)
        return result, calls, binary
    def admission_fixture(self):
        original = op.QA
        for name in ['freeze.json','RECIPE.md','SCRIPT_PLAN.md','PRIMARY_METADATA.json']:
            (self.root/name).write_bytes((original/name).read_bytes())
        (self.root/'recover-node22.py').write_text(SOURCE)
        (self.root/'LIVE_LAUNCHER.txt').write_text('fixed fake launcher data; not executed\n')
        sha = lambda name: hashlib.sha256((self.root/name).read_bytes()).hexdigest()
        data = {'schema':'SD-NODE22-ONE-ACQUISITION-1','sourceSha256':sha('recover-node22.py'),'launcherSha256':sha('LIVE_LAUNCHER.txt'),'recipeFreezeSha256':op.RECIPE_SHA,'pythonSha256':op.PYTHON_SHA,'reviews':[{'role':'Spec','reviewer':'/root/fake_spec','decision':'APPROVE_ONE_ACQUISITION_ONLY'},{'role':'Standards','reviewer':'/root/fake_standards','decision':'APPROVE_ONE_ACQUISITION_ONLY'}]}
        self.stack.enter_context(patch.object(op, 'QA', self.root))
        self.stack.enter_context(patch.object(sys, 'argv', [str(self.root/'recover-node22.py'),'--acquire-once']))
        self.stack.enter_context(patch.dict(os.environ, {}, clear=True))
        return data
    def save_admission(self, data):
        (self.root/'LIVE_ADMISSION.json').write_text(json.dumps(data))
    def test_exact_fake_admission_validates_without_acquiring(self):
        data = self.admission_fixture()
        self.save_admission(data)
        self.assertEqual(op.admit()['sourceSha256'], data['sourceSha256'])
        self.assertFalse((self.root/'ACQUISITION_ONCE.json').exists())
    def test_missing_admission_denies_without_claim(self):
        self.admission_fixture()
        with self.assertRaises(FileNotFoundError):
            op.admit()
        self.assertFalse((self.root/'ACQUISITION_ONCE.json').exists())
    def test_implementation_approval_cannot_acquire(self):
        data = self.admission_fixture()
        data['reviews'][0]['decision'] = 'APPROVE_NODE_RECOVERY_OPERATOR_IMPLEMENTATION_ONLY'
        self.save_admission(data)

            op.admit()
    def test_changed_source_binding_denied(self):

    def test_single_reviewer_denied(self):

    def test_main_consumed_claim_blocks_second_acquisition(self):
        fake = {'status':'TOOL_BYTES_RECOVERED_NOT_EXECUTED','cleanupExceptions':[]}
        with patch.object(op, 'QA', self.root), patch.object(op, 'admit', return_value={}), patch.object(op, 'recover', return_value=fake) as recover, patch.object(sys, 'argv', ['fake','--acquire-once']), contextlib.redirect_stdout(io.StringIO()):
            self.assertEqual(op.main(), 0)
            (self.root/'ACQUISITION_RESULT.json').unlink()
            with self.assertRaises(FileExistsError):
                op.main()
            self.assertEqual(recover.call_count, 1)
    def test_regular_file_replacement_not_removed(self):

        try:

                owned.remove()

if __name__ == '__main__':
    controls = []
    with patch.object(socket, 'socket', deny_network), patch.object(socket, 'create_connection', deny_network):
        result = unittest.TextTestRunner(verbosity=2).run(unittest.defaultTestLoader.loadTestsFromTestCase(OfflineChecks))
        baseline = op
        mutations = [
            ('HTTP_STATUS_GUARD_REMOVED', "need(response.status == 200, 'HTTP status refused; no redirect/retry')", 'pass', 'test_redirect_is_never_followed'),
            ('MEMBER_HASH_GUARD_REMOVED', "need(found and digest_fd(output_fd) == expected_binary, 'selected member missing/hash mismatch')", "need(found, 'selected member missing')", 'test_wrong_binary_digest_denied'),
            ('FILE_IDENTITY_GUARD_REMOVED', "need(identity(held) == identity(actual) == expected, 'file ownership drift')", 'pass', 'test_regular_file_replacement_not_removed'),
            ('OUTER_DEADLINE_RESET', 'min(seconds, old_remaining) if old_remaining else seconds', 'seconds', 'test_nested_deadline_does_not_reset_outer_budget'),
        ]
        for name, before, after, case in mutations:
            assert SOURCE.count(before) == 1, name
            op = load(SOURCE.replace(before, after))
            output = io.StringIO()
            outcome = unittest.TextTestRunner(stream=output, verbosity=2).run(OfflineChecks(case))
            controls.append({'mutation':name,'case':case,'testsRun':outcome.testsRun,'failures':len(outcome.failures),'errors':len(outcome.errors),'detected':not outcome.wasSuccessful(),'rawOutput':output.getvalue()})
        op = baseline
    assert real_network_attempts == [], 'Real networking was attempted'
    print(json.dumps({'status':'OFFLINE_CHECKS_COMPLETE','checks':result.testsRun,'failures':len(result.failures),'errors':len(result.errors),'skips':len(result.skipped),'controls':controls,'realNetworkAttempts':len(real_network_attempts),'NodeExecutions':0,'packageExecutions':0},indent=2))
    raise SystemExit(not (result.wasSuccessful() and all(row['detected'] for row in controls)))
