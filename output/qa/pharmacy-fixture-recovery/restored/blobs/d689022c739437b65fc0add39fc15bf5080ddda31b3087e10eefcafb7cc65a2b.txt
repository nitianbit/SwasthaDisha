const names = ['mongoose:cursor:next', 'mongoose:aggregate', 'mongoose:query', 'mongoose:model:save', 'mongoose:model:insertMany', 'mongoose:model:bulkWrite'];

function caught(action) { let failed = false; let value; try { action(); } catch (error) { failed = true; value = error; } assert(failed); return value; }
function harness(value, callback) {
  if (arguments.length === 0) value = 0;
  const d = fixture.createImportDiagnostics(); d.enter('PUBLIC_INITIALIZATION'); const effects = fixture.effectsState();
  let calls = 0; let argumentCount; let receiver; let traces = 0;
  const surface = { isBuildingSnapshot() { calls++; argumentCount = arguments.length; receiver = this; return callback ? callback() : value; }, tracingChannel() { traces++; return {}; } };
  const p = fixture.createPublicInitialization(surface, { publicInitialization: { tracingNames: names } }, effects, d);
  return { d, effects, p, surface, calls: () => calls, argumentCount: () => argumentCount, receiver: () => receiver, traces: () => traces };
}
function select(h) { return h.d.select('REPOSITORY_IMPORT', 'INVENTORY_EFFECTS'); }
test('primitive positive zero reaches actual snapshot callback unchanged', () => {
  const h = harness(); let returned;
  h.p.run(() => { returned = h.p.retrieve('v8').startupSnapshot.isBuildingSnapshot(); const dc = h.p.retrieve('node:diagnostics_channel'); for (const name of names) dc.tracingChannel(name); });
  assert(Object.is(returned, 0)); assert.equal(typeof returned, 'number'); assert.equal(h.calls(), 1); assert.equal(h.argumentCount(), 0); assert.equal(h.receiver(), h.surface);
  h.p.assertComplete(); fixture.assertEffects(h.effects); assert.equal(h.traces(), 6); assert.equal(select(h), 'INVENTORY_IMPORT_PUBLIC_INITIALIZATION_UNBRANDED');
});
