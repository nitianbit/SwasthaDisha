function createReadBudget(kind, clock) {
  need(['SNAPSHOT', 'METADATA'].includes(kind) && typeof clock === 'function', 'MEDICINE_CURSOR');
  const maxRows = kind === 'SNAPSHOT' ? 64 : 100; const maxNanos = 5000000000n;
  let previous = clock(); let elapsed = 0n; let paused = false; let failed = false; let rows = 0; let bytes = 0;
  need(typeof previous === 'bigint' && previous >= 0n, 'MEDICINE_CURSOR');
  function recordFailure() { if (!failed) { failed = true; lifetimeFailures++; } }
  function sample() {
    const value = clock(); need(typeof value === 'bigint' && value >= previous, 'MEDICINE_CURSOR');
    if (!paused) elapsed += value - previous; previous = value;
  }
  function check() {
    try { need(!failed && !paused, 'MEDICINE_CURSOR'); sample(); need(elapsed <= maxNanos, 'MEDICINE_CURSOR'); }
    catch (error) { recordFailure(); throw error; }
  }
  function account(rowCount, byteCount) {
    check(); rows += rowCount; bytes += byteCount;
    try { need(rows <= maxRows && bytes <= 128 * 1024, 'MEDICINE_CURSOR'); }
    catch (error) { recordFailure(); throw error; }
  }
  return Object.freeze({
    check, recordFailure,
    stats() { check(); return { millis: Number(elapsed) / 1000000, rows, bytes }; },
    beginCursor() { account(0, 2); },
    addRow(row, comma) { account(1, Buffer.byteLength(JSON.stringify(row)) + (comma ? 1 : 0)); },
    async creation(action) {
      check(); need(kind === 'METADATA' && typeof action === 'function', 'MEDICINE_CURSOR'); paused = true;
      try {
        return await withCleanup(action, () => { sample(); paused = false; check(); });
      } catch (error) { recordFailure(); throw error; }
    }
  });
}

async function collectReadCursor(makeCursor, limit, budget, leases) {
  const rows = []; let cursor; let first; let failed = false;
  const retain = error => { budget.recordFailure(); if (!failed) { failed = true; first = error; } };
  try {
    budget.check(); cursor = makeCursor(); budget.check();
    need(leases.has(cursor), 'MEDICINE_CURSOR'); budget.beginCursor();
    while (true) {
      budget.check(); const hasNext = await cursor.hasNext(); budget.check(); if (!hasNext) break;
      need(rows.length < limit, 'MEDICINE_CURSOR');
      budget.check(); const row = await cursor.next(); budget.check();
      need(plain(row), 'MEDICINE_CURSOR'); budget.addRow(row, rows.length > 0); rows.push(row);
    }
    need(cursor.id && typeof cursor.id.isZero === 'function' && cursor.id.isZero(), 'MEDICINE_CURSOR'); budget.check();
  } catch (error) { retain(error); }

  if (failed) throw first; return rows;
}

function canonical(value, bson, seen = new Set()) {
  if (value === null) return ['null'];
  if (value === undefined) return ['undefined'];
  if (typeof value === 'string') return ['string', value];
  if (typeof value === 'boolean') return ['boolean', value];
  if (typeof value === 'number') return ['number', Object.is(value, -0) ? '-0' : Number.isNaN(value) ? 'NaN' : value === Infinity ? '+Infinity' : value === -Infinity ? '-Infinity' : value];
  need(typeof value === 'object' && !seen.has(value), 'MEDICINE_TYPED_VALUE');
  if (value instanceof Date) {
    need(Object.getPrototypeOf(value) === Date.prototype && Number.isFinite(Date.prototype.getTime.call(value)) && Reflect.ownKeys(value).length === 0, 'MEDICINE_TYPED_VALUE');
    return ['Date', Date.prototype.getTime.call(value)];
  }
  if (value instanceof RegExp) {
    need(Object.getPrototypeOf(value) === RegExp.prototype && Reflect.ownKeys(value).length === 1
      && Object.getOwnPropertyDescriptor(value, 'lastIndex')?.value === 0, 'MEDICINE_TYPED_VALUE');
    return ['RegExp', Reflect.apply(Object.getOwnPropertyDescriptor(RegExp.prototype, 'source').get, value, []),
      Reflect.apply(Object.getOwnPropertyDescriptor(RegExp.prototype, 'flags').get, value, [])];
  }
  if (bson && value instanceof bson.ObjectId) {
    const hex = Reflect.apply(bson.ObjectId.prototype.toHexString, value, []);
    need(typeof hex === 'string' && /^[0-9a-f]{24}$/.test(hex), 'MEDICINE_TYPED_VALUE');
    return ['ObjectId', hex];
  }
  if (bson && (value instanceof bson.Timestamp || value instanceof bson.Long)) {
    const values = ['low', 'high', 'unsigned'].map(key => {
      const descriptor = Object.getOwnPropertyDescriptor(value, key);
      need(descriptor && own(descriptor, 'value'), 'MEDICINE_TYPED_VALUE'); return descriptor.value;
    });
    need(Number.isInteger(values[0]) && values[0] >= -2147483648 && values[0] <= 2147483647
      && Number.isInteger(values[1]) && values[1] >= -2147483648 && values[1] <= 2147483647
      && typeof values[2] === 'boolean', 'MEDICINE_TYPED_VALUE');
    return [value instanceof bson.Timestamp ? 'Timestamp' : 'Long', ...values];
  }
  if (bson && value instanceof bson.Binary) {
    const descriptors = ['buffer', 'position', 'sub_type'].map(key => Object.getOwnPropertyDescriptor(value, key));
    need(descriptors.every(item => item && own(item, 'value')), 'MEDICINE_TYPED_VALUE');
    const [buffer, position, subtype] = descriptors.map(item => item.value);
    need(buffer instanceof Uint8Array && integer(position) && position <= buffer.byteLength
      && integer(subtype) && subtype <= 255, 'MEDICINE_TYPED_VALUE');
    return ['Binary', subtype, position, Buffer.from(buffer.subarray(0, position)).toString('hex')];
  }
  seen.add(value);
  let result;
  if (Array.isArray(value)) {
    need(Object.getPrototypeOf(value) === Array.prototype && Reflect.ownKeys(value).length === value.length + 1
      && Array.from({ length: value.length }, (_, index) => Object.getOwnPropertyDescriptor(value, String(index)))
        .every(item => item && own(item, 'value') && item.enumerable), 'MEDICINE_TYPED_VALUE');
    result = ['array', value.map(item => canonical(item, bson, seen))];
  } else {
    const entries = canonicalDataEntries(value).sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0);
    need(entries.length === Reflect.ownKeys(value).length, 'MEDICINE_TYPED_VALUE');
    result = ['object', entries.map(([key, child]) => [key, canonical(child, bson, seen)])];
  }
  seen.delete(value); return result;
}

function canonicalRawCommand(command, bson) {
  const seen = new Set();
  function visit(value, location) {
    if (value instanceof Map) {
      need(command.createIndexes !== undefined && location === 'indexes.0.key', 'MEDICINE_COMMAND_ENVELOPE');
      return ['Map', indexKeyEntries(value).map(([key, item]) => [canonical(key, bson), canonical(item, bson)])];
    }
    if (Array.isArray(value)) {
      need(!seen.has(value) && Object.getPrototypeOf(value) === Array.prototype && Reflect.ownKeys(value).length === value.length + 1,
        'MEDICINE_TYPED_VALUE'); seen.add(value);
      const result = ['array', Array.from({ length: value.length }, (_, index) => {
        const descriptor = Object.getOwnPropertyDescriptor(value, String(index));
        need(descriptor && own(descriptor, 'value') && descriptor.enumerable, 'MEDICINE_TYPED_VALUE');
        return visit(descriptor.value, location ? `${location}.${index}` : String(index));
      })]; seen.delete(value); return result;
    }
    if (plain(value)) {
      need(!seen.has(value), 'MEDICINE_TYPED_VALUE'); seen.add(value);
      const entries = canonicalDataEntries(value).sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0);
      need(entries.length === Reflect.ownKeys(value).length, 'MEDICINE_TYPED_VALUE');
      const result = ['object', entries.map(([key, child]) => [key, visit(child, location ? `${location}.${key}` : key)])];
      seen.delete(value); return result;
    }
    return canonical(value, bson);
  }
  return visit(command, '');
}
