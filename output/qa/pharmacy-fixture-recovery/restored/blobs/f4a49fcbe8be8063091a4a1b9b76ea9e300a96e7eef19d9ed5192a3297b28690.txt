function createImportDiagnostics() {
  let current = null; let first = null; let closed = false;
  const invalid = () => { throw new Error('INVENTORY_DIAGNOSTIC'); };
  function record(category, decision) {
    if (current === null || closed || first !== null) return false;
    first = Object.freeze(decision === undefined ? { point: current, category } : { point: current, category, decision }); return true;
  }
  return Object.freeze({
    enter(point) {
      if (closed || !IMPORT_CHECKPOINTS.includes(point)) return invalid();
      current = point;
    },
    effect(category) {
      if (category === 'UNBRANDED' || !IMPORT_CATEGORIES.includes(category)) return invalid();
      return record(category);
    },
    guard(code) {
      if (!IMPORT_GUARD_CODES.includes(code)) return invalid();
      return record(Object.prototype.hasOwnProperty.call(IMPORT_GUARD_CATEGORIES, code) ? IMPORT_GUARD_CATEGORIES[code] : 'GUARD');
    },
    decision(reason) {
      if (typeof reason !== 'string' || !Object.prototype.hasOwnProperty.call(PUBLIC_INIT_DECISIONS, reason)) return invalid();
      return record('PUBLIC_INIT', PUBLIC_INIT_DECISIONS[reason]);
    },
    close() { closed = true; },
    select(finalStage, fallback) {
      if (!IMPORT_GUARD_CODES.includes(fallback)) return invalid();
      if (finalStage !== 'REPOSITORY_IMPORT') return fallback;
      if (first) return first.decision === undefined ? IMPORT_LOOKUP[first.point][first.category] : first.decision;
      return current === null ? fallback : IMPORT_LOOKUP[current].UNBRANDED;
    }
  });
}

function same(left, right, bson) {
  if (Object.is(left, right)) return true;
  if (left === null || right === null || typeof left !== typeof right || typeof left !== 'object') return false;
  if (left instanceof Date || right instanceof Date) return left instanceof Date && right instanceof Date && left.getTime() === right.getTime();

  if (bson) {
    for (const Type of [bson.ObjectId, bson.Timestamp, bson.Long, bson.Binary]) if (left instanceof Type || right instanceof Type) {
      return left instanceof Type && right instanceof Type && (Type === bson.Binary
        ? left.sub_type === right.sub_type && binaryBytes(left).equals(binaryBytes(right))
        : Type === bson.ObjectId ? left.toHexString() === right.toHexString()
          : left.low === right.low && left.high === right.high && left.unsigned === right.unsigned);
    }
  }
  if (left instanceof Map || right instanceof Map) return left instanceof Map && right instanceof Map
    && same([...left], [...right], bson);
  if (Array.isArray(left) || Array.isArray(right)) return Array.isArray(left) && Array.isArray(right)
    && left.length === right.length && left.every((value, i) => same(value, right[i], bson));

  return same(a, b, bson) && a.every(key => same(left[key], right[key], bson));
}

      if (++retrieved[name] !== 1) return deny(typeof name === 'string' && own(PUBLIC_INIT_RETRIEVE_REPEAT, name) ? PUBLIC_INIT_RETRIEVE_REPEAT[name] : undefined);
      return projections[name];
    },
    run(action) {
      const admitted = !open && !finished && typeof action === 'function';
      if (!admitted) diagnostic?.decision('RUN_ADMISSION');
      need(admitted, 'INVENTORY_PUBLIC_INITIALIZATION'); open = true;
      try { return action(); } finally { open = false; finished = true; }
    },
    assertComplete() {
      const complete = finished && !open && Object.values(retrieved).every(value => value === 1)
        && snapshotCalls === 1 && Object.values(traced).every(value => value === 1);
      if (!complete) diagnostic?.decision('COMPLETE');
      need(complete, 'INVENTORY_PUBLIC_INITIALIZATION');
    }
  });
}

function createReadBudget(kind, clock) {
  need(['SNAPSHOT', 'METADATA'].includes(kind) && typeof clock === 'function', 'INVENTORY_CURSOR');
  const maxRows = kind === 'SNAPSHOT' ? 64 : 100; const maxNanos = 5000000000n;
  let previous = clock(); let elapsed = 0n; let paused = false; let failed = false; let rows = 0; let bytes = 0;
  need(typeof previous === 'bigint' && previous >= 0n, 'INVENTORY_CURSOR');
  function recordFailure() { if (!failed) { failed = true; lifetimeFailures++; } }
  function sample() {
    const value = clock(); need(typeof value === 'bigint' && value >= previous, 'INVENTORY_CURSOR');
    if (!paused) elapsed += value - previous; previous = value;
  }
  function check() {
    try { need(!failed && !paused, 'INVENTORY_CURSOR'); sample(); need(elapsed <= maxNanos, 'INVENTORY_CURSOR'); }
    catch (error) { recordFailure(); throw error; }
  }
  function account(rowCount, byteCount) {
    check(); rows += rowCount; bytes += byteCount;
    try { need(rows <= maxRows && bytes <= 128 * 1024, 'INVENTORY_CURSOR'); }
    catch (error) { recordFailure(); throw error; }
  }
  return Object.freeze({
    check, recordFailure,
    beginCursor() { account(0, 2); },
    addRow(row, comma) { account(1, Buffer.byteLength(JSON.stringify(row)) + (comma ? 1 : 0)); },
    async creation(action) {
      check(); need(kind === 'METADATA' && typeof action === 'function', 'INVENTORY_CURSOR'); paused = true;
      try {
        return await withCleanup(action, () => { sample(); paused = false; check(); });
      } catch (error) { recordFailure(); throw error; }
    }
  });
}

async function collectReadCursor(makeCursor, limit, budget, leases) {
  const rows = []; let cursor; let first; let failed = false;
  const retain = error => { budget.recordFailure(); if (!failed) { failed = true; first = error; } };
  try {
    budget.check(); cursor = makeCursor(); budget.check();
    need(leases.has(cursor), 'INVENTORY_CURSOR'); budget.beginCursor();
    while (true) {
      budget.check(); const hasNext = await cursor.hasNext(); budget.check(); if (!hasNext) break;
      need(rows.length < limit, 'INVENTORY_CURSOR');
      budget.check(); const row = await cursor.next(); budget.check();
      need(plain(row), 'INVENTORY_CURSOR'); budget.addRow(row, rows.length > 0); rows.push(row);
    }
    need(!cursor.id || cursor.id.isZero(), 'INVENTORY_CURSOR'); budget.check();
  } catch (error) { retain(error); }
  try { budget.check(); } catch (error) { retain(error); }
  try { if (cursor && leases.has(cursor)) await cursor.close(); } catch (error) { retain(error); }
  try { budget.check(); } catch (error) { retain(error); }
  if (cursor) leases.delete(cursor);
  if (failed) throw first; return rows;
}

    && !/\bimport\s*\(|\bmodule\s*(?:\.|\[)|\.\s*(?:constructor|_compile|createRequire|binding|getBuiltinModule)\b/.test(source), 'INVENTORY_DOMAIN_AUDIT');
  for (const match of source.matchAll(/\brequire\s*\(([^)]*)\)/g)) need(/^\s*(['"])[^'"\r\n]+\1\s*$/.test(match[1]), 'INVENTORY_DOMAIN_AUDIT');
}

  }

  };

  }

  }

  }
  return Object.freeze({

    installTypeDi(typedi) {

      const service = typedi.Service;
      metadataArray = typedi.Container.globalInstance.services;
      need(Array.isArray(metadataArray) && metadataArray.length === 0, 'INVENTORY_REPOSITORY_FRESHNESS'); metadataBase = metadataArray.length;
      // Service imports this exact cached sentinel; an arbitrary same-named
      // Symbol must never stand in for its uninstantiated metadata value.
      const emptyModule = require.cache[path.join(ROOT, 'node_modules/typedi/cjs/empty.const.js')];
      const emptyDescriptor = Object.getOwnPropertyDescriptor(emptyModule?.exports || {}, 'EMPTY_VALUE');
      need(emptyDescriptor && own(emptyDescriptor, 'value') && typeof emptyDescriptor.value === 'symbol', 'INVENTORY_REPOSITORY_FRESHNESS');
      const emptyValue = emptyDescriptor.value;
      installTypeDiSetBoundary(typedi, () => decoratingTarget, emptyValue);
      // The public export is a getter. Instrument its actual cached writable
      // decorator export, rather than replacing TypeDI with a fake root object.
      const serviceFilename = path.join(ROOT, 'node_modules/typedi/cjs/decorators/service.decorator.js');
      const serviceModule = require.cache[serviceFilename];
      const descriptor = Object.getOwnPropertyDescriptor(serviceModule?.exports || {}, 'Service');
      need(descriptor && own(descriptor, 'value') && descriptor.value === service && descriptor.writable, 'INVENTORY_REPOSITORY_FRESHNESS');
      serviceModule.exports.Service = function (...args) {
        need(args.length === 0, 'INVENTORY_REPOSITORY_FRESHNESS'); const decorate = service.apply(this, args);
        return target => {
          need(typeof target === 'function' && ports.serviceRegistrations.includes(target.name) && registrations < 51
            && !registeredClasses.includes(target), 'INVENTORY_REPOSITORY_FRESHNESS');
          registrations++; let result; decoratingTarget = target;
          try { result = decorate(target); } finally { decoratingTarget = null; }
          registeredClasses.push(target);
          const actual = metadataArray[metadataArray.length - 1];
          need(metadataArray.length === metadataBase + registrations && actual?.id === target && actual.type === target
            && actual.eager === false && actual.global === false && actual.transient === false && actual.multiple === false
            && actual.factory === undefined && actual.value === emptyValue, 'INVENTORY_REPOSITORY_FRESHNESS'); return result;
        };
      };
      methods.push(() => Object.defineProperty(serviceModule.exports, 'Service', descriptor));
      for (const name of ['get', 'getMany', 'reset', 'remove', 'of', 'import', 'initialize', 'registerHandler']) {
        if (typeof typedi.Container[name] === 'function') typedi.Container[name] = () => loadDeny();
        if (typeof typedi.ContainerInstance.prototype[name] === 'function') typedi.ContainerInstance.prototype[name] = () => loadDeny();
      }
    },
    observeRepositories(action) { need(!observer && typeof action === 'function', 'INVENTORY_CONTEXT'); observer = action; },
    bindModels(actual) { need(!models && keys(actual, MODEL_NAMES), 'INVENTORY_MODEL'); models = actual; },
    prepareRepositories() {
      need(!heldCache && models && types && targets.every(filename => !own(require.cache, filename)), 'INVENTORY_REPOSITORY_FRESHNESS');
      heldCache = new Map(Object.entries(require.cache)); children = module.children; heldChildren = children.slice();
    },
    repository(kind, variantId) {
      need(heldCache && !restored, 'INVENTORY_REPOSITORY_FRESHNESS');
      if (kind === 'UNSAFE' && nextVariant === 0) {
        // The initial 139 nominal cases may exercise already admitted lazy
        // public modules. Preserve all prior identities, then freeze this set
        // before actual-source mutation/restoration trials.
        for (const [filename, value] of heldCache) if (!targets.includes(filename)) need(require.cache[filename] === value, 'INVENTORY_REPOSITORY_FRESHNESS');
        for (const [filename, value] of Object.entries(require.cache)) if (!heldCache.has(filename) && !targets.includes(filename)) {
          need(relative(filename).startsWith('node_modules/') && own(manifest.files, relative(filename)), 'INVENTORY_REPOSITORY_FRESHNESS');
          heldCache.set(filename, value);
        }
      }
      stableCache(); clearRepository();
      let filename; let digest;
      if (kind === 'NOMINAL_INITIAL') {
        need(evaluations === 0 && variantId === null, 'INVENTORY_REPOSITORY_FRESHNESS'); filename = targets[0]; digest = ports.sourceFiles[REPOSITORY_SOURCE];
      } else if (kind === 'UNSAFE') {
        need(evaluations === 1 + nextVariant * 2 && pendingRestore === null && variantId === ports.requiredSensitivityIds[nextVariant], 'INVENTORY_REPOSITORY_FRESHNESS');
        pendingRestore = variantId; filename = path.join(ROOT, plan.mutants[variantId].file); digest = plan.mutants[variantId].mutantSha256;
      } else {
        need(kind === 'RESTORE' && pendingRestore === variantId && evaluations === 2 + nextVariant * 2, 'INVENTORY_REPOSITORY_FRESHNESS');
        pendingRestore = null; nextVariant++; filename = targets[0]; digest = ports.sourceFiles[REPOSITORY_SOURCE];
      }
      const before = registrations; const value = require(filename); const exported = exportsOf(value); evaluations++;
      need(registrations === before + 3 && registrations === evaluations * 3, 'INVENTORY_REPOSITORY_FRESHNESS');
      if (evaluations === 1) { originalRepository = require.cache[filename]; finalChildren = children.slice(); }
      need(observer, 'INVENTORY_CONTEXT'); observer(exported[0]);
      stableCache();
      return Object.freeze({ Item: exported[0], Adjustment: exported[1], Reservation: exported[2], evaluation: evaluations, sourceSha256: digest, models });
    },
    restoreRepository() {
      if (restored) return;
      let first; let failed = false;
      try { stableCache(); } catch (error) { failed = true; first = error; }
      try {
        clearRepository();
        for (const [filename, value] of heldCache) if (targets.includes(filename)) require.cache[filename] = value;
        if (originalRepository) require.cache[targets[0]] = originalRepository;
        children.splice(0, children.length, ...(finalChildren || heldChildren));
        stableCache();
        need(targets.slice(1).every(filename => !own(require.cache, filename))
          && (!originalRepository || require.cache[targets[0]] === originalRepository), 'INVENTORY_REPOSITORY_FRESHNESS');
        for (const restore of methods.reverse()) restore();
        restored = true;
      } catch (error) { if (!failed) { failed = true; first = error; } }
      if (failed) throw first;
    },
    audit: () => ({ evaluations, registrations, restored }),
    assertHealthy() {
      need(Module._load === load && Module._resolveFilename === resolve && effects.loaderDenials === 0, 'INVENTORY_LOADER');
      need(reached.size === 5 && DOMAIN_FILES.every(filename => reached.has(filename)), 'INVENTORY_MODEL');
    },
    recordMethodRestore: action => methods.push(action)
  });
}

function createWireBoundary(manifest, ports, cases, bson, effects) {

  const totals = { hello: 0, buildinfo: 0, create: 0, createIndexes: 0, listIndexes: 0, listCollections: 0 };
  const actionCounts = new Map(); const createdCollections = new Set(); const builtIndexes = new Set();

  function sid(value) {
    need(keys(value, ['id']) && value.id instanceof bson.Binary && value.id.sub_type === 4

  function metadata(command) {

    if (own(command, '$clusterTime')) {
      const value = command.$clusterTime;
      need(keys(value, ['clusterTime', 'signature']) && value.clusterTime instanceof bson.Timestamp
        && keys(value.signature, ['hash', 'keyId']) && value.signature.hash instanceof bson.Binary
        && value.signature.hash.sub_type === 0 && binaryBytes(value.signature.hash).length === 20
        && (value.signature.keyId === 0 || (isLong(value.signature.keyId) && value.signature.keyId.isZero())), 'INVENTORY_COMMAND_ENVELOPE');
    }
  }
  const common = ['$db', 'lsid', '$clusterTime', '$readPreference', 'txnNumber', 'autocommit', 'startTransaction', 'readConcern'];
  const commandKeys = {
    hello: ['hello'], buildinfo: ['buildinfo'], create: ['create'],
    createIndexes: ['createIndexes', 'indexes'], listIndexes: ['listIndexes', 'cursor', 'maxTimeMS'],
    listCollections: ['listCollections', 'filter', 'cursor', 'nameOnly', 'authorizedCollections', 'maxTimeMS'],
    find: ['find', 'filter', 'limit', 'singleBatch', 'batchSize', 'maxTimeMS'],
    update: ['update', 'updates', 'ordered'], insert: ['insert', 'documents', 'ordered'], delete: ['delete', 'deletes', 'ordered'],
    commitTransaction: ['commitTransaction', 'writeConcern', 'maxTimeMS'], abortTransaction: ['abortTransaction', 'writeConcern'],
    getMore: ['getMore', 'collection', 'batchSize', 'maxTimeMS'], killCursors: ['killCursors', 'cursors'],
    endSessions: ['endSessions', 'writeConcern']
  };
  function boundedEvent(value) {
    const item = clone(value, bson); eventBytes += Buffer.byteLength(JSON.stringify(item));
    need(events.length < 16000 && eventBytes <= 64 * 1024 * 1024, 'INVENTORY_COMMAND_LIMIT'); events.push(freezeRecord(item)); return item;
  }
  function cursorOwner(command, name, sessionId) {
    const id = long(name === 'getMore' ? command.getMore : command.cursors?.[0]); const held = cursors.get(id);
    need(held && held.database === manifest.database && held.session === sessionId && held.collection === (name === 'getMore' ? command.collection : command.killCursors), 'INVENTORY_CURSOR');
    return held;
  }
  function actionCounter() {
    need(scope && scope.row && scope.sessionId, 'INVENTORY_PHASE');
    let count = actionCounts.get(scope.token);
    if (!count) { count = { find: 0, write: 0, terminal: 0, failureAbort: 0 }; actionCounts.set(scope.token, count); }
    return count;
  }
  function checkStarted(event) {
    const { commandName: name, command, databaseName } = event;
    need(own(commandKeys, name) && plain(command) && command.$db === databaseName
      && Object.keys(command).every(key => commandKeys[name].includes(key) || common.includes(key)), 'INVENTORY_COMMAND_ENVELOPE'); metadata(command);
    const collection = command[name];
    if (name === 'endSessions') {
      need(phase === 'CLOSE' && databaseName === 'admin' && ++endSessions === 1
        && Array.isArray(collection) && collection.length > 0 && collection.length <= 256
        && collection.every(value => ownedSessionIds.has(sid(value)))
        && (!own(command, 'writeConcern') || same(command.writeConcern, { w: 0 }))
        && !own(command, 'txnNumber') && !own(command, 'autocommit'), 'INVENTORY_CLOSE');
      return;
    }
    const sessionId = sid(command.lsid); need(ownedSessionIds.has(sessionId), 'INVENTORY_SESSION');
    if (['commitTransaction', 'abortTransaction'].includes(name)) {
      need(scope && databaseName === 'admin' && sessionId === scope.sessionId && command[name] === 1
        && command.autocommit === false && isLong(command.txnNumber) && scope.transaction === true
        && same(command.txnNumber, scope.txnNumber, bson)
        && !own(command, 'startTransaction') && !own(command, 'readConcern')
        && same(command.writeConcern, { w: 'majority', wtimeout: 5000 }), 'INVENTORY_TERMINAL');

      if (phase === 'CONTROL') need(scope.terminal === (name === 'commitTransaction' ? 'COMMIT' : 'ABORT')
        && ++count.terminal === 1 && (name !== 'commitTransaction' || command.maxTimeMS === 5000), 'INVENTORY_TERMINAL');
      else need(phase === 'CASE_CLEANUP' && scope.terminal === 'FAILURE_ABORT' && name === 'abortTransaction'

      cursorOwner(command, name, sessionId); return;
    }

    }
    need(scope && scope.row, 'INVENTORY_PHASE');
    if (phase === 'ACTION') {
      need(collection === 'inventoryitems' && sessionId === scope.sessionId && ['find', 'insert', 'update'].includes(name), 'INVENTORY_COMMANDS');

          && command.limit === 1 && command.singleBatch === true && !own(command, 'batchSize'), 'INVENTORY_COMMANDS');
      } else {
        need(++count.write <= expected.write.inventoryitems && command.ordered === true, 'INVENTORY_COMMANDS');
        if (name === 'update') {
          need(Array.isArray(command.updates) && command.updates.length === 1, 'INVENTORY_COMMANDS'); const item = command.updates[0];
          need(keys(item, ['q', 'u'], ['multi', 'upsert']) && (!own(item, 'multi') || item.multi === false)
            && (!own(item, 'upsert') || item.upsert === false) && same(item.q, { _id: scope.fixture.ids.targetItem, __v: 0 }, bson)
            && keys(item.u, ['$set', '$inc']) && same(item.u.$inc, { __v: 1 }) && plain(item.u.$set)
            && Array.isArray(item.u.$set.batches) && item.u.$set.batches.length >= 1 && item.u.$set.batches.length <= 3
            && Object.keys(item.u.$set).every(key => ['batches', 'availableQty', 'reservedQty', 'isActive', 'purchasePrice',
              'retailPrice', 'wholesalePrice', 'reorderThreshold', 'rackLocation', 'updatedAt'].includes(key)), 'INVENTORY_COMMANDS');
        } else {
          need(scope.fixture.newItem && cases.metadataCaseIds.NEW_ITEM_INSERT_V0.includes(scope.row.id)
            && Array.isArray(command.documents) && command.documents.length === 1, 'INVENTORY_COMMANDS');
          const item = command.documents[0]; const id = oid(item._id);
          need(!scope.fixture.generated.has(id) && same(item.pharmacyId, scope.fixture.ids.pharmacy, bson)
            && same(item.medicineId, scope.fixture.ids.medicine, bson), 'INVENTORY_OWNERSHIP_PROOF'); scope.inserted = clone(item, bson);
        }
      }
      if (scope.transaction) {
        need(command.autocommit === false && isLong(command.txnNumber), 'INVENTORY_TRANSACTION');
        if (!scope.txnNumber) { need(name === 'find' && command.startTransaction === true && same(command.readConcern, { level: 'snapshot' }), 'INVENTORY_TRANSACTION'); scope.txnNumber = clone(command.txnNumber, bson); }
        else need(same(scope.txnNumber, command.txnNumber, bson) && !own(command, 'startTransaction') && !own(command, 'writeConcern'), 'INVENTORY_TRANSACTION');
      } else need(!own(command, 'txnNumber') && !own(command, 'autocommit') && !own(command, 'startTransaction') && !own(command, 'readConcern'), 'INVENTORY_TRANSACTION');

    }
    if (name === 'find') {
      need(COLLECTIONS.includes(collection) && same(command.filter, {}) && command.batchSize === 100 && command.maxTimeMS === 5000
        && !own(command, 'limit') && !own(command, 'singleBatch') && (sessionId === controlId || (phase === 'CONTROL' && scope.transaction && sessionId === scope.sessionId)), 'INVENTORY_COMMANDS'); return;
    }
    if (phase === 'CASE_SETUP') {
      need(name === 'insert' && COLLECTIONS.includes(collection) && sessionId === controlId && command.ordered === true
        && Array.isArray(command.documents) && command.documents.length === scope.fixture.seed[collection].length
        && command.documents.every(item => scope.seedIds[collection].has(oid(item._id))), 'INVENTORY_OWNERSHIP_PROOF'); return;
    }
    need(phase === 'CASE_CLEANUP' && name === 'delete' && COLLECTIONS.includes(collection) && sessionId === controlId
      && command.ordered === true && Array.isArray(command.deletes) && command.deletes.length === 1
      && keys(command.deletes[0], ['q', 'limit']) && command.deletes[0].limit === 0
      && same(command.deletes[0].q, { _id: { $in: scope.allowedCleanup[collection] } }, bson), 'INVENTORY_CLEANUP');
  }
  return Object.freeze({
    ownSession(value, control = false) { const id = sid(value); ownedSessionIds.add(id); if (control) { need(!controlId, 'INVENTORY_SESSION'); controlId = id; } return id; },
    phase(next, nextScope) { need(['CONNECT', 'INDEX_SETUP', 'CASE_SETUP', 'ACTION', 'CONTROL', 'CASE_CLEANUP', 'CLOSE'].includes(next), 'INVENTORY_PHASE'); phase = next; scope = nextScope; },
    started(event) {

        need(Number.isSafeInteger(event.requestId) && !pending.has(event.requestId), 'INVENTORY_COMMANDS'); checkStarted(event);

      } catch (error) { effects.unexpectedCommands++; throw error; }
    },
    succeeded(event) {
      try {
        const start = pending.get(event.requestId); need(start && start.commandName === event.commandName && plain(event.reply)
          && event.reply.ok === 1 && !own(event.reply, 'writeConcernError') && !own(event.reply, 'writeErrors'), 'INVENTORY_COMMAND_FAILURE');
        pending.delete(event.requestId);
        if (event.reply.cursor) {

            || keys(cursor, ['id', 'ns', 'nextBatch']), 'INVENTORY_CURSOR');
          const id = apmCursorZero(cursor.id); const command = start.command;
          const continuation = start.commandName === 'getMore' ? cursorOwner(command, 'getMore', sid(command.lsid)) : null;
          const owner = continuation || start.lease;
          const collection = start.commandName === 'listCollections' ? '$cmd.listCollections' : continuation ? continuation.collection : command[start.commandName];
          const ns = start.databaseName + '.' + collection;
          need(owner && cursor.ns === ns && Array.isArray(cursor.firstBatch || cursor.nextBatch), 'INVENTORY_CURSOR');
          const batch = cursor.firstBatch || cursor.nextBatch; owner.rows += batch.length; owner.bytes += Buffer.byteLength(JSON.stringify(batch));
          need(owner.rows <= owner.rowLimit && owner.bytes <= 128 * 1024, 'INVENTORY_CURSOR');
          if (continuation) cursors.delete(long(command.getMore));
          if (id !== '0') {
            need(!cursors.has(id), 'INVENTORY_CURSOR'); cursors.set(id, owner);
          }
        }
        boundedEvent({ kind: 'succeeded', requestId: event.requestId, commandName: event.commandName, reply: event.reply });
      } catch (error) { effects.commandFailures++; throw error; }
    },
    failed(event) {
      const start = pending.get(event.requestId); effects.commandFailures++;
      if (start) { pending.delete(event.requestId); boundedEvent({ kind: 'failed', requestId: event.requestId, commandName: event.commandName }); }
      fail('INVENTORY_COMMAND_FAILURE');
    },
    events: () => events.slice(),
    totals: () => ({ ...totals }),
    expectCursor(owner) { need(owner && owner.cursor && !leases.some(value => value.cursor === owner.cursor), 'INVENTORY_CURSOR'); leases.push({ ...owner, started: false, rows: 0, bytes: 0 }); },
    closeCursor(id) { cursors.delete(long(id)); },
    assertHealthy() { need(effects.unexpectedCommands === 0 && effects.commandFailures === 0 && pending.size === 0, 'INVENTORY_COMMAND_FAILURE'); },
    assertClosed() { need(cursors.size === 0 && pending.size === 0, 'INVENTORY_CLOSE'); },
    reject, sid, oid
  });
}

      need(same(afterCollections.slice().sort(), COLLECTIONS.slice().sort()) && actual.create === 3 && actual.createIndexes === 22

        commands: { create: actual.create, createIndexes: actual.createIndexes, listIndexes: actual.listIndexes, listCollections: actual.listCollections } };
      wire.assertHealthy(); budget.check(); return indexProof;
    },
    observeRepository(Repository) {

        const original = Object.getOwnPropertyDescriptor(Repository.prototype, name); need(original && typeof original.value === 'function', 'INVENTORY_REPOSITORY_FRESHNESS');
        Object.defineProperty(Repository.prototype, name, { ...original, value: async function (...args) {
          need(phase === 'ACTION' && scope && scope.row.method === name && this.constructor === Repository && ++methodCalls <= 163
            && args.length === 4 && args[3] === actionSession, 'INVENTORY_REPOSITORY_FRESHNESS');
          const result = await original.value.apply(this, args);
          if (result && scope.fixture.newItem) {

          }

        } });
        loader.recordMethodRestore(() => Object.defineProperty(Repository.prototype, name, original));
      }
    },
    phase: switchPhase,
    async startSession() {
      need(phase === 'CONTROL' && scope && scope.setupCollections.size === 3 && (!actionSession || actionSession.hasEnded)
        && ++actionSessions <= 163, 'INVENTORY_SESSION');
      sessionRequest = 'ACTION'; try { actionSession = client.startSession({ causalConsistency: false }); } finally { sessionRequest = null; }
      need(actionSession !== control && sessionIdentity(actionSession) !== sessionIdentity(control), 'INVENTORY_SESSION'); return actionSession;
    },
    async snapshot(session) {
      const budget = createReadBudget('SNAPSHOT', readClock);
      need(['CASE_SETUP', 'CONTROL', 'CASE_CLEANUP'].includes(phase) && scope && (session === control || (phase === 'CONTROL' && scope.transaction && session === actionSession)), 'INVENTORY_SESSION');
      const result = {}; let count = 0; let bytesSeen = 0;
      for (const collection of COLLECTIONS) {
        result[collection] = await array(() => mongoose.connection.db.collection(collection).find({}, { session, batchSize: 100, maxTimeMS: 5000 }), 16, budget);
        count += result[collection].length; bytesSeen += Buffer.byteLength(JSON.stringify(result[collection]));
      }
      need(count <= 64 && bytesSeen <= 128 * 1024, 'INVENTORY_CURSOR'); wire.assertHealthy();
      if (phase === 'CASE_SETUP' && !scope.emptyProved) { need(count === 0, 'INVENTORY_OWNERSHIP_PROOF'); scope.emptyProved = true; }
      if (phase === 'CASE_CLEANUP') { need(count === 0, 'INVENTORY_CLEANUP'); scope.finalEmptyProved = true; }
      if (phase === 'CONTROL' && scope.returnedNewId && scope.inserted) {
        const item = result.inventoryitems.find(row => wire.oid(row._id) === scope.returnedNewId);
        if (item) {
          need(wire.oid(scope.inserted._id) === scope.returnedNewId && same(item.pharmacyId, scope.fixture.ids.pharmacy, bson)
            && same(item.medicineId, scope.fixture.ids.medicine, bson), 'INVENTORY_OWNERSHIP_PROOF'); scope.observedNewId = scope.returnedNewId;
        }
      }
      budget.check(); return result;
    },
    get models() { return models; }, get controlSession() { return control; }, get indexProof() { return indexProof; },
    assertHealthy() {
      assertEffects(effects); wire.assertHealthy(); seedBulk.assertIdle();
      need(mongoose.connections.length === 1 && same(mongoose.modelNames().sort(), MODEL_NAMES.slice().sort())
        && MODEL_NAMES.every(name => mongoose.models[name] === models[name]), 'INVENTORY_MODEL');
      need(leases.size === 0 && lifetimeFailures === 0, 'INVENTORY_CURSOR');
    },
    assertCompleted() { seedBulk.assertIdle(); need(methodCalls === 163 && actionSessions === 163 && transactions === 5 && scope?.finalEmptyProved && actionSession.hasEnded, 'INVENTORY_RESULT'); },
    async close() {
      need(!closeStarted, 'INVENTORY_CLOSE'); closeStarted = true; phase = 'CLOSE'; wire.phase(phase, scope);
      let first; let failed = false;
      const attempt = async action => { try { await action(); } catch (error) { if (!failed) { failed = true; first = error; } } };
      for (const cursor of leases) await attempt(async () => { await cursor.close(); leases.delete(cursor); });
      for (const [session, record] of sessions) if (!record.ended && record.kind !== 'IMPLICIT') await attempt(async () => {
        need(!session.inTransaction(), 'INVENTORY_CLOSE'); await session.endSession();
      });
      if (client) await attempt(() => mongoose.disconnect());
      nativeBoundary.mongoClosed();
      await attempt(() => { wire.assertClosed(); assertEffects(effects); need(leases.size === 0 && [...sessions].every(([session]) => session.hasEnded), 'INVENTORY_CLOSE'); });
      if (failed) throw first;
    }
  });
}

function denseArray(value, maximum) {
  return Array.isArray(value) && value.length <= maximum && Reflect.ownKeys(value).length === value.length + 1
    && Object.keys(value).length === value.length && Object.keys(value).every((key, i) => key === String(i)
      && own(Object.getOwnPropertyDescriptor(value, key), 'value'));
}
function validateCleanupIds(eligible, proof, oid) {
  need(keys(eligible, COLLECTIONS), 'INVENTORY_CLEANUP'); const allowed = {};
  for (const collection of COLLECTIONS) {
    const values = eligible[collection]; need(denseArray(values, 3), 'INVENTORY_CLEANUP'); const ids = new Set();
    for (const id of values) {
      const key = oid(id); need(!ids.has(key), 'INVENTORY_CLEANUP'); ids.add(key);
      need(proof.emptyProved && (proof.seedIds[collection].has(key)
        || (collection === 'inventoryitems' && proof.returnedNewId === key && proof.observedNewId === key && proof.inserted
          && oid(proof.inserted._id) === key)), 'INVENTORY_OWNERSHIP_PROOF');
    }
    allowed[collection] = values.slice();
  }
  return allowed;
}
function createSeedBulkLease() {
  let active = null;
  function scope(collection, options, control) {
    need(active && collection === active.collection && options?.session === control && options.ordered !== false, 'INVENTORY_COMMANDS');
  }
  return Object.freeze({
    async run(collection, documents, action) {
      need(!active && collection && denseArray(documents, 2) && documents.length > 0 && typeof action === 'function', 'INVENTORY_COMMANDS');
      active = { collection, documents: documents.slice() };
      try { return await action(); } finally { active = null; }
    },
    assertBulk(collection, operations, options, control) {
      scope(collection, options, control);
      need(denseArray(operations, 2) && operations.length === active.documents.length
        && operations.every((operation, i) => keys(operation, ['insertOne']) && keys(operation.insertOne, ['document'])
          && operation.insertOne.document === active.documents[i]), 'INVENTORY_COMMANDS');
    },
    assertOrdered: scope,
    assertIdle() { need(active === null, 'INVENTORY_COMMANDS'); }
  });
}
function guardOwnedClientConstructor(Constructor, admitted, uri, options, projection, capture) {
  let calls = 0; let guarded;
  guarded = new Proxy(Constructor, {
    construct(target, args, newTarget) {
      need(admitted() === true && ++calls === 1 && newTarget === guarded && args.length === 2 && args[0] === uri, 'INVENTORY_DATABASE');
      const supplied = args[1];
      need(plain(supplied) && keys(supplied.runtimeAdapters, ['os'])
        && keys(supplied.runtimeAdapters.os, ['platform', 'arch', 'release', 'type'])
        && Object.keys(projection).every(name => supplied.runtimeAdapters.os[name] === projection[name])
        && Object.entries(options).filter(([name]) => !['autoCreate', 'autoIndex'].includes(name)).every(([name, value]) => same(supplied[name], value)), 'INVENTORY_OS_ADAPTER');
      const instance = Reflect.construct(target, args, target); capture(instance); return instance;
    },
    apply() { fail('INVENTORY_DATABASE'); }
  });
  return guarded;
}
function returnedNewItemId(result, method, fixture, bson) {
  const item = method === 'adjustBatchStock' ? result?.item : result;
  need(plain(item) && item._id instanceof bson.ObjectId && /^[a-f0-9]{24}$/.test(item._id.toHexString())
    && !fixture.generated.has(item._id.toHexString()) && same(item.pharmacyId, fixture.ids.pharmacy, bson)
    && same(item.medicineId, fixture.ids.medicine, bson), 'INVENTORY_OWNERSHIP_PROOF');
  // Actual repository methods return saved.toObject(), never a hydrated Doc.
  return item._id.toHexString();
}
function inspectManifest(manifest, plan, ports, cases) {
  const observed = {}; let total = 0;
  function walk(directory, prefix = '') {
    for (const name of fs.readdirSync(directory)) {
      const relative = prefix + name; const filename = path.join(directory, name); const metadata = fs.lstatSync(filename);
      need(!metadata.isSymbolicLink(), 'INVENTORY_FILE_MANIFEST');
      if (metadata.isDirectory()) walk(filename, relative + '/');
      else {
        need(metadata.isFile() && metadata.nlink === 1, 'INVENTORY_FILE_MANIFEST');
        if (relative === '.admission.json') continue;
        total += metadata.size; need(total <= 256 * 1024 * 1024 && Object.keys(observed).length < 2777, 'INVENTORY_FILE_MANIFEST');
        observed[relative] = fs.readFileSync(filename);
      }
    }
  }
  walk(ROOT); return validateAdmissionData(manifest, plan, ports, cases, { runId: process.env.APP_RUN_ID, files: observed });
}
