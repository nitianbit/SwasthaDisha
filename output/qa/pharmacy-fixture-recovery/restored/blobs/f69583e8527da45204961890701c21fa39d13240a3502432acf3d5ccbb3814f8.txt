    const virtual = { filename: path.join(runtimeRoot, selected.sourceId), sourceId: selected.sourceId };
    const row = { sourceId: selected.sourceId, sourceSha256: sha(Buffer.from(selected.text)), status: 'EXCEPTION',
      programFiles: [], libraryFiles: {}, diagnostics: [] };
    // Every other host operation is exactly the corrected broad host operation.
    // Only these two fixed in-memory virtual source names/texts are served here.
    const probeHost = { ...host,
      fileExists: file => path.resolve(file) === virtual.filename || host.fileExists(file),
      readFile: file => path.resolve(file) === virtual.filename ? selected.text : host.readFile(file),
      getSourceFile(file, languageVersion, onError, shouldCreateNewSourceFile) {
        if (path.resolve(file) === virtual.filename) return ts.createSourceFile(file, selected.text, languageVersion, true);
        return host.getSourceFile(file, languageVersion, onError, shouldCreateNewSourceFile);
      },
    };
    try {
      const program = ts.createProgram([virtual.filename], options, probeHost);
      const actual = ts.getPreEmitDiagnostics(program);
      requireThat(actual.length <= 256, 'BROAD_LIBRARY_DIAGNOSTIC_LIMIT');
      row.diagnostics = actual.map(item => metadata(item, virtual));
      const observed = inspect(program, virtual); row.programFiles = observed.programFiles; row.libraryFiles = observed.libraryFiles;
      row.status = 'COMPLETE';
    } catch { /* A compiler exception remains EXCEPTION, never a detected control. */ }
    return row;
  }
  function sameLibraries(row) {
    return JSON.stringify(Object.keys(row.libraryFiles).sort()) === JSON.stringify(BROAD_DEFAULT_LIBRARY_NAMES)
      && BROAD_DEFAULT_LIBRARY_NAMES.every(name => row.libraryFiles[name] === BROAD_LIBRARY_HASHES[name]);
  }
  function positivePredicate(row) {
    const sourceId = BROAD_POSITIVE_SOURCE.sourceId;
    return row.status === 'COMPLETE' && row.diagnostics.length === 0 && sameLibraries(row)
      && JSON.stringify(row.programFiles) === JSON.stringify([...libraryIds, sourceId].sort());
  }
  function guarded(action) { try { return action(); } catch { return fail(); } }
  return Object.freeze({
    read(file, inheritedRead) {
      const name = libraryName(file);
      return name === null ? inheritedRead(file) : admittedLibrary(file);
    },
    createProgram(action) {
      return guarded(() => {
        requireThat(ts.version === '5.9.3' && sha(regular(path.join(libraryRoot, 'typescript.js'))) === INDEX_COMPILER_SHA,
          'BROAD_LIBRARY_COMPILER');
        const program = action(); report.semanticProgram = inspect(program); return program;
      });
    },
    getDiagnostics: action => guarded(action),
    check(host, inheritedGetDefaultLibLocation) {
      return guarded(() => {
        const correctedLocation = host.getDefaultLibLocation;
        requireThat(typeof correctedLocation === 'function' && typeof inheritedGetDefaultLibLocation === 'function'
          && correctedLocation() === libraryRoot, 'BROAD_LIBRARY_HOST_LOCATION');
        report.positive = probe(host, BROAD_POSITIVE_SOURCE);
        requireThat(positivePredicate(report.positive), 'BROAD_LIBRARY_POSITIVE');
        report.negative = probe(host, BROAD_NEGATIVE_SOURCE);
        const negative = report.negative; const diagnostic = negative.diagnostics[0];
        requireThat(negative.status === 'COMPLETE' && negative.diagnostics.length === 1 && sameLibraries(negative)
          && JSON.stringify(negative.programFiles) === JSON.stringify([...libraryIds, BROAD_NEGATIVE_SOURCE.sourceId].sort())
          && diagnostic.sourceId === BROAD_NEGATIVE_SOURCE.sourceId && diagnostic.code === 2322 && diagnostic.category === 1
          && diagnostic.start === 13 && diagnostic.length === 12, 'BROAD_LIBRARY_NEGATIVE');
        try { host.getDefaultLibLocation = inheritedGetDefaultLibLocation; report.oldHostControl = probe(host, BROAD_POSITIVE_SOURCE); }
        finally { host.getDefaultLibLocation = correctedLocation; }
        requireThat(host.getDefaultLibLocation === correctedLocation && report.oldHostControl.status === 'COMPLETE'
          && !positivePredicate(report.oldHostControl) && !sameLibraries(report.oldHostControl)
          && report.oldHostControl.diagnostics.some(diagnostic => diagnostic.code === 2318 && diagnostic.message === "Cannot find global type 'Array'."),
        'BROAD_LIBRARY_PRIOR_HOST_CONTROL');
        report.restoredPositive = probe(host, BROAD_POSITIVE_SOURCE);
        requireThat(positivePredicate(report.restoredPositive), 'BROAD_LIBRARY_RESTORED_POSITIVE');
        report.status = 'PASS'; return report;
      });
    },
  });
}


function assertModelDeclarations(declarations, models) {
  const select = rows => rows.map(row => ({ name: row.name, source: row.source }))
    .sort((a, b) => a.name.localeCompare(b.name));
  requireThat(Array.isArray(declarations) && Array.isArray(models) && models.length === 3
    && new Set(models.map(row => row.name)).size === models.length, 'DUPLICATE_DECLARED_MODEL');
  requireThat(new Set(declarations.map(row => row.name)).size === declarations.length, 'DUPLICATE_EMITTED_MODEL');
  requireThat(JSON.stringify(select(declarations)) === JSON.stringify(select(models)), 'EMITTED_MODEL_MANIFEST_MISMATCH');
}

/** Emitted CJS is parsed as data only; no application module is required. */
function inspectEmittedModels(runtimeRoot, modules, ports) {
  const reachable = new Set(); const declarations = []; const queue = [modules[ports.runtimeRoot]];
  const sourceFor = file => 'src1/' + file.slice('compiled/'.length, -3) + '.ts';
  const unwrap = expression => {
    while (ts.isParenthesizedExpression(expression)) expression = expression.expression;
    if (ts.isBinaryExpression(expression) && expression.operatorToken.kind === ts.SyntaxKind.CommaToken) return unwrap(expression.right);
    return expression;
  };
  function requiredName(node) {
    if (ts.isCallExpression(node) && ts.isIdentifier(node.expression) && node.expression.text === 'require') {
      requireThat(node.arguments.length === 1 && ts.isStringLiteralLike(node.arguments[0]), 'DYNAMIC_DOMAIN_REQUIRE');
      return node.arguments[0].text;
    }
    if (ts.isCallExpression(node) && ts.isIdentifier(node.expression)
      && ['__importDefault', '__importStar'].includes(node.expression.text) && node.arguments.length === 1)
      return requiredName(node.arguments[0]);
    return undefined;
  }
  while (queue.length) {
    const file = queue.shift(); if (reachable.has(file)) continue;
    requireThat(typeof file === 'string' && file.startsWith('compiled/') && !file.startsWith('compiled/mutants/'), 'MODEL_AUDIT_SOURCE');
    reachable.add(file);
    const text = fs.readFileSync(path.join(runtimeRoot, file), 'utf8');
    const ast = ts.createSourceFile(file, text, ts.ScriptTarget.ES2020, true, ts.ScriptKind.JS);
    auditDomain(ast, true);
    const mongooseBindings = new Set();
    function bindings(node) {
      if (ts.isVariableDeclaration(node) && ts.isIdentifier(node.name) && node.initializer
        && requiredName(node.initializer) === 'mongoose') mongooseBindings.add(node.name.text);
      ts.forEachChild(node, bindings);
    }
    bindings(ast);
    function visit(node) {
      if (ts.isCallExpression(node)) {
        const request = requiredName(node);
        if (request) {
          if (request.startsWith('@/')) { requireThat(modules[request], 'MODEL_AUDIT_UNKNOWN_ALIAS'); queue.push(modules[request]); }
          else if (request.startsWith('.')) {
            const base = path.posix.normalize(path.posix.join(path.posix.dirname(file), request));
            const target = [base + '.js', base + '/index.js'].find(candidate => Object.values(modules).includes(candidate));
            requireThat(target, 'MODEL_AUDIT_UNKNOWN_RELATIVE'); queue.push(target);
          } else requireThat(ports.publicRuntimeRoots.includes(request), 'MODEL_AUDIT_UNKNOWN_PACKAGE');
        }
        const expression = unwrap(node.expression);
        if (ts.isPropertyAccessExpression(expression) && expression.name.text === 'model' && node.arguments.length >= 2) {
          let receiver = expression.expression;
          if (ts.isPropertyAccessExpression(receiver) && receiver.name.text === 'default') receiver = receiver.expression;
          if (ts.isIdentifier(receiver) && mongooseBindings.has(receiver.text)) {
            requireThat(ts.isStringLiteralLike(node.arguments[0]), 'DYNAMIC_MODEL_DECLARATION');
            declarations.push({ name: node.arguments[0].text, source: sourceFor(file), compiled: file });
          }
        }
      }
      ts.forEachChild(node, visit);
    }
    visit(ast);
  }
  declarations.sort((a, b) => a.name.localeCompare(b.name));
  assertModelDeclarations(declarations, ports.models);
  requireThat(declarations.length === 3 && reachable.size === 5, 'INVENTORY_RUNTIME_MODEL_CLOSURE');
  const controls = [];
  const expectRejected = (id, candidate) => {
    let rejected = false; try { assertModelDeclarations(declarations, candidate); } catch { rejected = true; }
    requireThat(rejected, 'MODEL_AUDIT_CONTROL_SURVIVED'); controls.push({ id, rejected: true });
  };
  expectRejected('OMITTED_ADJUSTMENT_MODEL', ports.models.filter(row => row.name !== 'InventoryAdjustment'));
  expectRejected('EXTRA_UNREACHABLE_MODEL', [...ports.models, { name: 'UnreachableFixtureModel', source: 'src1/unreachable.ts' }]);
  expectRejected('DUPLICATE_MODEL', [...ports.models, ports.models[0]]);
  expectRejected('WRONG_MODEL_SOURCE', ports.models.map((row, index) => index === 0 ? { ...row, source: 'src1/wrong.ts' } : row));
  expectRejected('OMITTED_RESERVATION_MODEL', ports.models.filter(row => row.name !== 'InventoryReservation'));
  requireThat(JSON.stringify(controls.map(row => row.id)) === JSON.stringify(MODEL_CONTROLS), 'MODEL_AUDIT_CONTROL_SET');
  return { declarations, audit: { status: 'STATIC_EMITTED_MODEL_MAP_VERIFIED', reachableModules: reachable.size,
    modelCount: declarations.length, controls, applicationEvaluations: 0 } };
}

const domainAuditFailures = new WeakMap();
const DOMAIN_IDENTIFIERS = new Set(['eval', 'Function', 'process', 'global', 'globalThis', '__non_webpack_require__']);
const DOMAIN_PROPERTIES = new Set(['getBuiltinModule', 'createRequire', 'mainModule', 'binding', 'dlopen', '_load', '_resolveFilename', 'constructor']);
function domainNeed(ok, code) {
  if (!ok) { const error = new Error(code); domainAuditFailures.set(error, code); throw error; }
}
function domainAuditFailureCode(error) {
  return error !== null && (typeof error === 'object' || typeof error === 'function') ? domainAuditFailures.get(error) ?? null : null;
}

/** Pure policy used by the real source/emitted AST walker; never loads TypeScript. */
function assertDomainPolicyFact(fact) {
  const fields = { IMPORT: ['kind'], IDENTIFIER: ['kind', 'name', 'emitted', 'directLiteralCall'],
    PROPERTY: ['kind', 'name'], ELEMENT_LITERAL: ['kind', 'name'] };
  domainNeed(fact !== null && typeof fact === 'object' && !Array.isArray(fact)
    && Object.hasOwn(fields, fact.kind)
    && JSON.stringify(Object.keys(fact).sort()) === JSON.stringify([...fields[fact.kind]].sort()), 'DOMAIN_NATIVE_OR_CODEGEN');
  if (fact.kind === 'IMPORT') return domainNeed(false, 'DYNAMIC_DOMAIN_REQUIRE');
  domainNeed(typeof fact.name === 'string', 'DOMAIN_NATIVE_OR_CODEGEN');
  if (fact.kind === 'IDENTIFIER') {
    domainNeed(typeof fact.emitted === 'boolean' && typeof fact.directLiteralCall === 'boolean', 'DOMAIN_NATIVE_OR_CODEGEN');
    domainNeed(!DOMAIN_IDENTIFIERS.has(fact.name), 'DOMAIN_NATIVE_OR_CODEGEN');
    domainNeed(fact.name !== 'require' || (fact.emitted === true && fact.directLiteralCall === true), 'DYNAMIC_DOMAIN_REQUIRE');
  } else domainNeed(!DOMAIN_PROPERTIES.has(fact.name), 'DOMAIN_NATIVE_OR_CODEGEN');
}

/** Reject native/codegen properties and require aliases in BOTH actual AST walks. */
function auditDomain(ast, emitted) {
  function visit(node) {
    if (node.kind === ts.SyntaxKind.ImportKeyword) assertDomainPolicyFact({ kind: 'IMPORT' });
    if (ts.isIdentifier(node)) {
      const parent = node.parent;
      const directLiteralCall = Boolean(parent && ts.isCallExpression(parent) && parent.expression === node
        && parent.arguments.length === 1 && ts.isStringLiteralLike(parent.arguments[0]));
      assertDomainPolicyFact({ kind: 'IDENTIFIER', name: node.text, emitted, directLiteralCall });
    }
    if (ts.isPropertyAccessExpression(node)) assertDomainPolicyFact({ kind: 'PROPERTY', name: node.name.text });
    if (ts.isElementAccessExpression(node) && ts.isStringLiteralLike(node.argumentExpression))
      assertDomainPolicyFact({ kind: 'ELEMENT_LITERAL', name: node.argumentExpression.text });
    ts.forEachChild(node, visit);
  }
  visit(ast);
}

/** Future admitted emit only: parse fixed virtual text as data, never evaluate it. */
function checkDomainAuditControls(sourceCommit) {
  const controls = [];
  for (const fixture of DOMAIN_AUDIT_FIXTURES) {
    requireThat(sha(Buffer.from(fixture.text)) === fixture.fixtureSha256, 'DOMAIN_AUDIT_CONTROL_HASH');
    const ast = ts.createSourceFile(fixture.id, fixture.text, ts.ScriptTarget.ES2020, true,
      fixture.format === 'TS' ? ts.ScriptKind.TS : ts.ScriptKind.JS);
    requireThat(ast.parseDiagnostics.length === 0, 'DOMAIN_AUDIT_CONTROL_PARSE');
    let observed = 'ACCEPT';
    try { auditDomain(ast, fixture.format === 'CJS'); }
    catch (error) {
      const known = domainAuditFailureCode(error);
      requireThat(known === 'DOMAIN_NATIVE_OR_CODEGEN' || known === 'DYNAMIC_DOMAIN_REQUIRE', 'DOMAIN_AUDIT_CONTROL_UNEXPECTED_FAILURE');
      observed = known;
    }
    requireThat(observed === fixture.expected, 'DOMAIN_AUDIT_CONTROL_SURVIVED');
    controls.push({ id: fixture.id, format: fixture.format, fixtureSha256: fixture.fixtureSha256, expected: fixture.expected, observed });
  }
  requireThat(controls.length === 21, 'DOMAIN_AUDIT_CONTROL_COUNT');
  process.stdout.write(JSON.stringify({ schema: 'SD-INVENTORY-DOMAIN-AUDIT-CONTROLS-1', compiler: ts.version,
    sourceCommit, applicationEvaluations: 0, controls }) + '\n');
}

function compile(sourceRoot, runtimeRoot, inputPlan) {
  requireThat(ts === undefined, 'COMPILER_SINGLE_INVOCATION');
  const portsBytes = fs.readFileSync(path.join(runtimeRoot, 'ports-manifest.json'));
  const casesBytes = fs.readFileSync(path.join(runtimeRoot, 'cases.json'));
  requireThat(sha(portsBytes) === PORTS_SHA && sha(casesBytes) === CASES_SHA
    && inputPlan.portsSha === PORTS_SHA && inputPlan.casesSha === CASES_SHA, 'COMPILER_FROZEN_DATA');
  const ports = JSON.parse(portsBytes); const cases = JSON.parse(casesBytes);
  requireThat(ports.schema === 'SD-INVENTORY-QUALITY-PORTS-1' && ports.role === 'inventory-quality-repository'
    && Object.keys(ports.sourceFiles).length === 7 && cases.cases.length === 139 && cases.sourceMutations.length === 8, 'COMPILER_SCOPE');
  ts = require('typescript');
  requireThat(ts.version === '5.9.3', 'COMPILER_VERSION');
  checkDomainAuditControls(inputPlan.sourceCommit);
  const tracked = new Set(inputPlan.trackedSources); const modules = {}; const sources = {}; const diagnostics = [];
  function resolveAlias(request, from) {
    let relative;
    if (request.startsWith('@/')) relative = 'src1/' + request.slice(2);
    else if (request.startsWith('.')) relative = path.posix.normalize(path.posix.join(path.posix.dirname(from), request));
    else return null;
    requireThat(relative.startsWith('src1/') && !relative.includes('..'), 'SOURCE_ESCAPE');
    const found = [relative + '.ts', relative + '/index.ts'].find(name => Object.hasOwn(ports.sourceFiles, name));
    requireThat(found && tracked.has(found), 'UNRESOLVED_CONSERVATIVE_SOURCE'); return found;
  }
  function emit(relative, text, output) {
    const result = ts.transpileModule(text, { fileName: relative, reportDiagnostics: true,
      compilerOptions: { target: ts.ScriptTarget.ES2020, module: ts.ModuleKind.CommonJS,
        experimentalDecorators: true, emitDecoratorMetadata: true, esModuleInterop: true, importHelpers: false, sourceMap: false } });
    for (const item of result.diagnostics || []) diagnostics.push({ file: relative, code: item.code, category: item.category,
      message: ts.flattenDiagnosticMessageText(item.messageText, '\n') });
    requireThat((result.diagnostics || []).length === 0, 'TRANSPILATION_DIAGNOSTIC');
    const ast = ts.createSourceFile(output, result.outputText, ts.ScriptTarget.ES2020, true, ts.ScriptKind.JS); auditDomain(ast, true);
    fs.mkdirSync(path.dirname(path.join(runtimeRoot, output)), { recursive: true }); fs.writeFileSync(path.join(runtimeRoot, output), result.outputText);
  }
  for (const [relative, expected] of Object.entries(ports.sourceFiles)) {
    requireThat(tracked.has(relative), 'UNTRACKED_CONSERVATIVE_SOURCE');
    const absolute = path.join(sourceRoot, relative); const metadata = fs.lstatSync(absolute);
    requireThat(metadata.isFile() && !metadata.isSymbolicLink() && metadata.nlink === 1 && fs.realpathSync(absolute) === absolute, 'NONREGULAR_SOURCE');
    const bytes = fs.readFileSync(absolute); const text = bytes.toString('utf8');
    requireThat(sha(bytes) === expected, 'CONSERVATIVE_SOURCE_HASH'); sources[relative] = expected;
    const ast = ts.createSourceFile(relative, text, ts.ScriptTarget.ES2020, true); auditDomain(ast, false);
    for (const statement of ast.statements) {
      if ((ts.isImportDeclaration(statement) || ts.isExportDeclaration(statement)) && statement.moduleSpecifier) {
        requireThat(ts.isStringLiteralLike(statement.moduleSpecifier), 'DYNAMIC_DOMAIN_MODULE'); const request = statement.moduleSpecifier.text;
        if (request.startsWith('@/') || request.startsWith('.')) resolveAlias(request, relative);
        else requireThat(ports.publicRuntimeRoots.includes(request), 'UNREVIEWED_DOMAIN_PACKAGE');
      }
    }
    const output = 'compiled/' + relative.slice(5, -3) + '.js';
    modules['@/' + relative.slice(5, -3).replace(/\/index$/, '')] = output; modules['@/' + relative.slice(5, -3)] = output;
    emit(relative, text, output);
  }
  requireThat(Object.keys(sources).length === 7 && new Set(Object.values(modules)).size === 7, 'CONSERVATIVE_SOURCE_COUNT');
  const modelMap = inspectEmittedModels(runtimeRoot, modules, ports); const mutants = {};
  for (const variant of cases.sourceMutations) {
    requireThat(variant.source === 'src1/modules/pharmacyInventory/repository/index.ts'
      && Object.hasOwn(sources, variant.source), 'MUTANT_SOURCE_NOT_ADMITTED');
    const original = fs.readFileSync(path.join(sourceRoot, variant.source), 'utf8');
    requireThat(sha(Buffer.from(original)) === sources[variant.source], 'MUTANT_ORIGINAL_HASH');
    const changed = once(original, variant.before, variant.after);
    requireThat(changed !== original && sha(Buffer.from(changed)) === variant.predictedSourceSha256, 'MUTANT_SOURCE_HASH');
    const output = 'compiled/mutants/' + variant.id + '.js'; emit(variant.source, changed, output);
    mutants[variant.id] = { source: variant.source, originalSha256: sources[variant.source], mutantSha256: variant.predictedSourceSha256, file: output };
  }
  requireThat(JSON.stringify(Object.keys(mutants)) === JSON.stringify(ports.requiredSensitivityIds), 'MUTANT_SET');
  const options = { target: ts.ScriptTarget.ES2020, module: ts.ModuleKind.CommonJS, moduleResolution: ts.ModuleResolutionKind.Node10,
    experimentalDecorators: true, emitDecoratorMetadata: true, esModuleInterop: true, noEmit: true, skipLibCheck: true, types: [] };
  const host = ts.createCompilerHost(options); const fileExists = host.fileExists; const readFile = host.readFile;
  const sourcePaths = new Set(Object.keys(sources).map(name => path.join(sourceRoot, name)));
  const typeRoot = path.join(runtimeRoot, 'node_modules') + path.sep; const libraryRoot = path.join(typeRoot, 'typescript/lib');
  const inheritedGetDefaultLibLocation = host.getDefaultLibLocation;
  const libraryAdmission = createBroadLibraryAdmission(sourceRoot, runtimeRoot, options);
  const allowedTypeFile = file => sourcePaths.has(path.resolve(file)) || path.resolve(file).startsWith(typeRoot);
  host.fileExists = file => allowedTypeFile(file) && fileExists(file);
  host.readFile = file => allowedTypeFile(file) ? libraryAdmission.read(file, readFile) : undefined;
  host.getSourceFile = (file, languageVersion, onError) => {
    const text = host.readFile(file); if (text === undefined) { if (onError) onError('DECLARED_TYPE_INPUT_UNAVAILABLE'); return undefined; }
    return ts.createSourceFile(file, text, languageVersion, true);
  };
  host.getCurrentDirectory = () => runtimeRoot;
  host.getDefaultLibFileName = () => path.join(typeRoot, 'typescript/lib/lib.es2020.full.d.ts'); host.getDefaultLibLocation = () => libraryRoot;
  host.resolveModuleNames = (names, containingFile) => names.map(request => {
    if (sourcePaths.has(containingFile) && (request.startsWith('@/') || request.startsWith('.'))) {
      const relative = resolveAlias(request, path.relative(sourceRoot, containingFile).split(path.sep).join('/'));
      const absolute = relative && path.join(sourceRoot, relative); requireThat(sourcePaths.has(absolute), 'SEMANTIC_SOURCE_OUTSIDE_CLOSURE');
      return { resolvedFileName: absolute, extension: ts.Extension.Ts, isExternalLibraryImport: false };
    }
    const from = sourcePaths.has(containingFile) ? path.join(runtimeRoot, '__public_types__.ts') : containingFile;
    return ts.resolveModuleName(request, from, options, host).resolvedModule;
  });
  const program = libraryAdmission.createProgram(() => ts.createProgram([...sourcePaths], options, host));
  const semanticDiagnostics = libraryAdmission.getDiagnostics(() => ts.getPreEmitDiagnostics(program)).map(item => ({
    file: item.file ? path.relative(sourceRoot, item.file.fileName).split(path.sep).join('/') : null,
    code: item.code, category: item.category, message: ts.flattenDiagnosticMessageText(item.messageText, '\n') }));
  const broadLibraryReport = libraryAdmission.check(host, inheritedGetDefaultLibLocation);
  const plan = { schema: 'SD-INVENTORY-QUALITY-PLAN-1', role: ports.role, sourceCommit: inputPlan.sourceCommit, sourceFiles: sources,
    modules, mutants, models: ports.models, modelDeclarations: modelMap.declarations, modelMapAudit: modelMap.audit,
    packageVersions: inputPlan.packageVersions, lockSha: inputPlan.lockSha, portsSha: inputPlan.portsSha, casesSha: inputPlan.casesSha,
    requiredCaseIds: ports.requiredCaseIds, requiredGroupMethodCounts: ports.requiredGroupMethodCounts,
    requiredSensitivityIds: ports.requiredSensitivityIds, providerCounterKeys: ports.providerCounterKeys,
    providerCounterScope: 'ALL_PHASES_INCLUDING_SENSITIVITY_ZERO', nominalMethodCalls: 139, unsafeMethodCalls: 8, restorationMethodCalls: 16,
    compilation: { status: 'TRANSPILE_ONLY_NOT_TYPECHECKED', compiler: ts.version, diagnostics, semanticDiagnostics,
      semanticScope: 'DECLARED_SEVEN_SOURCE_CLOSURE_AND_PUBLIC_TYPES', fullBuild: false, libraryAdmission: broadLibraryReport } };
  fs.writeFileSync(path.join(runtimeRoot, 'runtime-plan.json'), JSON.stringify(plan, null, 2) + '\n'); return plan;
}
